## 一、接口：AgentMemory（`capabilities/agent_memory/base.py`）

```python
class AgentMemory(ABC):
    """Agent 记忆接口——存/查工具用法 + 文本记忆"""

    # ── 存：记录一次工具用法 ──
    @abstractmethod
    async def save_tool_usage(
        self, question, tool_name, args, context, success=True, metadata=None
    ) -> None:
        """保存一个工具用法模式（供未来参考）"""
        pass

    # ── 存：文本记忆 ──
    @abstractmethod
    async def save_text_memory(self, content, context) -> TextMemory:
        """保存自由文本记忆（洞察/观察/上下文）"""
        pass

    # ── 查：找类似的工具用法 ──
    @abstractmethod
    async def search_similar_usage(
        self, question, context, *, limit=10, similarity_threshold=0.7, tool_name_filter=None
    ) -> List[ToolMemorySearchResult]:
        """根据问题找类似的工具用法模式"""
        pass

    # ── 查：文本记忆 ──
    @abstractmethod
    async def search_text_memories(self, query, context, *, limit=10, similarity_threshold=0.7):
        """搜索存储的文本记忆"""
        pass

    # ── 其他：最近记忆 / 删除 / 清空 ──
    async def get_recent_memories(self, context, limit=10): ...
    async def delete_by_id(self, context, memory_id) -> bool: ...
    async def clear_memories(self, context, tool_name=None): ...
```

**核心 4 方法**：存（save_tool_usage）+ 存文本（save_text_memory）+ 查（search_similar_usage）+ 查文本（search_text_memories）。

## 二、真实实现：DemoAgentMemory（`integrations/local/agent_memory/in_memory.py`）逐段

> 零依赖内存实现——不装向量库，用 Jaccard + difflib 算相似度。展示记忆怎么存、怎么查。

### 数据结构

```python
class DemoAgentMemory(AgentMemory):
    def __init__(self, *, max_items: int = 10_000):
        self._memories: List[ToolMemory] = []        # 工具用法记忆
        self._text_memories: List[TextMemory] = []   # 文本记忆
        self._lock = asyncio.Lock()                  # 并发安全
        self._max_items = max_items                  # 上限（FIFO 淘汰）
```

### 存：save_tool_usage（registry 怎么记住）

```python
async def save_tool_usage(self, question, tool_name, args, context, success=True, metadata=None):
    """保存一个工具用法模式"""
    tm = ToolMemory(
        memory_id=str(uuid.uuid4()),   # 唯一 ID
        question=question,             # 用户问什么
        tool_name=tool_name,           # 用了哪个工具
        args=args,                     # 传了什么参数（如 SQL）
        timestamp=self._now_iso(),     # 时间戳
        success=success,               # 是否成功
        metadata=metadata or {},
    )
    async with self._lock:             # 并发安全
        self._memories.append(tm)
        # FIFO 淘汰：超过上限删最早的
        if len(self._memories) > self._max_items:
            overflow = len(self._memories) - self._max_items
            del self._memories[:overflow]
```

**关键**：存的是"问题 + 工具 + 参数 + 是否成功"——这就是成功查询模式。

### 相似度算法（Jaccard + difflib）

```python
@staticmethod
def _similarity(a, b) -> float:
    """两个字符串的相似度（取 Jaccard 和 difflib 的最大值）"""
    a_norm, b_norm = cls._normalize(a), cls._normalize(b)   # 小写+压缩空格

    # Jaccard：词集合交集/并集
    ta, tb = cls._tokenize(a_norm), cls._tokenize(b_norm)   # 按空格分词
    jaccard = len(ta & tb) / len(ta | tb) if (ta | tb) else 1.0

    # difflib：字符序列相似度
    ratio = difflib.SequenceMatcher(None, a_norm, b_norm).ratio()

    return max(jaccard, ratio)   # 取两者最大值
```

**关键**：**不装向量库**，用 Jaccard（词重叠）+ difflib（字符相似）算相似度——够 demo 用，生产换向量库。

### 查：search_similar_usage（下次怎么复用）

```python
async def search_similar_usage(self, question, context, *, limit=10, similarity_threshold=0.7, tool_name_filter=None):
    """找类似的历史成功用法"""
    q = self._normalize(question)

    async with self._lock:
        # ① 候选：只看成功的 + 可选按工具过滤
        candidates = [
            m for m in self._memories
            if m.success and (tool_name_filter is None or m.tool_name == tool_name_filter)
        ]

        # ② 打分：每个候选和当前问题算相似度
        results = []
        for m in candidates:
            score = self._similarity(q, m.question)
            results.append((m, min(score, 1.0)))

        # ③ 过滤阈值 + 排序
        results = [(m, s) for (m, s) in results if s >= similarity_threshold]
        results.sort(key=lambda x: x[1], reverse=True)

        # ④ 返回前 limit 个（带相似度分数）
        ...
```

**关键**：**下次用户问类似问题 → 搜记忆 → 命中（相似度≥0.7）→ 复用之前的 SQL 模式**。这就是"训练式 RAG"——不是现查，是记住复用。

## 三、记忆工具（`tools/agent_memory.py`）—— LLM 可主动调

```python
# 工具 1：记住成功模式（LLM 调它）
class SaveQuestionToolArgsTool(Tool[SaveQuestionToolArgsParams]):
    name = "save_question_tool_args"
    description = "Save a successful question-tool-argument combination for future reference"
    async def execute(self, context, args):
        await context.agent_memory.save_tool_usage(   # ★ 写记忆
            question=args.question, tool_name=args.tool_name,
            args=args.args, context=context, success=True,
        )
        return ToolResult(success=True, result_for_llm="Saved to memory")

# 工具 2：找成功模式（LLM 调它）
class SearchSavedCorrectToolUsesTool(Tool[SearchSavedCorrectToolUsesParams]):
    name = "search_saved_correct_tool_uses"
    description = "Search for similar tool usage patterns based on a question"
    async def execute(self, context, args):
        results = await context.agent_memory.search_similar_usage(   # ★ 读记忆
            question=args.question, context=context,
            limit=args.limit, similarity_threshold=args.similarity_threshold,
        )
        return ToolResult(success=True, result_for_llm=format_results(results))
```

**关键**：这两个工具注册进 ToolRegistry → LLM 在循环里能主动调：
- 执行工具前 → 调 search（查有没有成功先例）
- 执行成功后 → 调 save（记下来）

## 四、记忆在循环里的完整流程（训练式 RAG 怎么运作）

```
用户问"上个月逾期率"
  → LLM 先调 search_saved_correct_tool_uses("上个月逾期率")
      → 记忆里有类似成功模式？→ 返回之前的 SQL
  → LLM 复用 SQL → 调 run_sql 执行
  → 成功 → LLM 调 save_question_tool_args(question, "run_sql", {sql})
      → 存进记忆（下次还能用）
```

**这就是训练式 RAG**：
- 不是每次从零生成 SQL
- 先查记忆 → 有类似先例就复用 → 成功后更新记忆
- **越用越聪明**（记忆累积）

## 五、对照 ai_assistant 的改造动作

你的 RAG 是"静态检索文档"，记忆是"动态学习成功查询"。改造：

```python
# 现在：静态 RAG（每次现查文档）
def _handle_definition_query(self, query):
    docs = retriever.search(query)   # 每次现查
    return explain(docs)

# 改造：加成功查询记忆（训练式）
class CreditAgentMemory:
    def save_usage(self, question, sql, success):
        # 记住：这个问题 + 这个 SQL 成功了
        self._memories.append({question, sql, success})

    def search_usage(self, question):
        # 找类似成功 SQL
        similar = [m for m in self._memories if similarity(question, m.question) > 0.7]
        return similar

# 在 pipeline 里：查数前先搜记忆，命中复用
cached_sql = memory.search_usage(query)
if cached_sql:
    result = executor.execute(cached_sql)   # 复用，不重新生成
else:
    sql = sql_generator.generate(...)       # 重新生成
    result = executor.execute(sql)
    if result.success:
        memory.save_usage(query, sql)       # 记住
```

**改造要点**：
1. 加 `CreditAgentMemory`（存/查成功 SQL）
2. 查数前先搜记忆 → 命中复用
3. 成功后存记忆 → 越用越聪明
